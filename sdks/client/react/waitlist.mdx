---
title: "Waitlist"
description: "Waitlist signup form"
---

A mutation, not a data hook - `join()` returns a promise, same shape as [`useCheckout`](/sdks/client/react/billing/checkout-and-portal).

```tsx
import { useWaitlist } from '@kerne/react';

function WaitlistForm() {
  const { join } = useWaitlist();
  const [email, setEmail] = useState('');
  const [submitted, setSubmitted] = useState(false);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    await join({ email });
    setSubmitted(true);
  };

  if (submitted) return <p>You're on the list.</p>;
  return (
    <form onSubmit={handleSubmit}>
      <input value={email} onChange={(e) => setEmail(e.target.value)} type="email" required />
      <button type="submit">Join waitlist</button>
    </form>
  );
}
```

`join()` also accepts `name` and arbitrary `metadata`. See [registration modes](/concepts/identity/registration-modes) for how waitlist mode fits into the broader signup flow, and the [server SDK](/sdks/server/waitlist/join) for listing/removing entries (admin-only, not exposed here).
