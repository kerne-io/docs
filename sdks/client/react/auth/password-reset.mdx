---
title: "Password"
description: "Change or reset a password"
---

## Change password (logged in)

Exposed directly on `useAuth()` - no server round-trip of your own needed:

```tsx
import { useAuth } from '@kerne/react';

function ChangePasswordForm() {
  const { updatePassword } = useAuth();

  await updatePassword({ currentPassword, newPassword });
}
```

## Forgot password (logged out)

```tsx
import { useAuth } from '@kerne/react';

function ForgotPasswordForm() {
  const { requestPasswordReset } = useAuth();

  await requestPasswordReset(email); // sends the reset email
}
```

## Reset password (from the emailed link)

Read the token from wherever your reset link lands (a query param, typically), then confirm:

```tsx
import { useAuth } from '@kerne/react';

function ResetPasswordForm({ token }: { token: string }) {
  const { confirmPasswordReset } = useAuth();

  await confirmPasswordReset(token, newPassword);
}
```
